]

def compute_period(form, report_date_str):
    report_date = datetime.strptime(report_date_str, "%Y-%m-%d")
    if form == "10-K":
        period_end = report_date
        period_start = report_date.replace(year=report_date.year - 1)
    else:  # 10-Q
        period_end = report_date
        period_start = report_date - timedelta(days=89)  # approximate quarter start
    return period_start.strftime("%Y-%m-%d"), period_end.strftime("%Y-%m-%d")

        "Filing Date": f["filingDate"],
        "Accession Number": f["accessionNumber"],
        "SEC URL": f["url"]
    })

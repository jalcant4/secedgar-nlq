import datetime

results = []

    end = datetime.date.fromisoformat(f['reportDate'])
    if f['form'] == '10-K':
        start = end.replace(year=end.year - 1)
    else:  # 10-Q
        # 3 months prior
        month = end.month - 3

        'accessionNumber': f['accessionNumber'],

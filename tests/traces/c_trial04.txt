# Compute fiscal period starts and ends for the 4 filings
from datetime import datetime, timedelta

filings = relevant  # from state
results = []
for f in filings:
    report_date = datetime.strptime(f['reportDate'], '%Y-%m-%d')
    url = filing_url(cik, f['accessionNumber'], f['primaryDocument'])
    if f['form'] == '10-Q':
        period_start = (report_date - timedelta(days=90)).strftime('%Y-%m-%d')
    else:  # 10-K
        period_start = (report_date - timedelta(days=365)).strftime('%Y-%m-%d')
    period_end = f['reportDate']
    results.append({
        'form': f['form'],
        'accessionNumber': f['accessionNumber'],
        'filingDate': f['filingDate'],
        'periodStart': period_start,
        'periodEnd': period_end,
        'url': url
    })
for r in results:
    print(r)

# Parse period starts from filing texts
period_starts = []
for f in k_q_filings[:4]:
    t = texts[f['accessionNumber']]
    # Find "for the three months ended" or similar
    m = re.search(r'for the three months? ended ([A-Za-z]+ \d{1,2}, \d{4})', t, re.IGNORECASE)
    if m:
        start_str = m.group(1)
        # For 10-Q, period end is reportDate, period start is ~3 months prior
        # Parse the quarter start
        from datetime import datetime, timedelta
        report_dt = datetime.strptime(f['reportDate'], '%Y-%m-%d')
        if report_dt.month in [3, 6, 9, 12]:
            # Quarter end: month 3->Jan 1, 6->Apr 1, 9->Jul 1, 12->Oct 1
            quarter_starts = {3: '01-01', 6: '04-01', 9: '07-01', 12: '10-01'}
            start_dt = datetime.strptime(f"{report_dt.year}-{quarter_starts[report_dt.month]}-01", '%Y-%m-%d')
        else:
            start_dt = None
        period_starts.append(start_dt.strftime('%Y-%m-%d') if start_dt else 'N/A')
